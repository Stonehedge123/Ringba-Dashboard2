📝 Setup Instructions:

Step 1: Open this HTML file on a computer/monitor that you want to display alerts

Step 2: In your Google Apps Script, add this function and call it when you detect issues:

// For Hangup Alerts
function sendHangupAlert(targetName, hangupRate, callCount) {
  UrlFetchApp.fetch('http://localhost:8888/alert', {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify({
      type: 'hangup',
      target: targetName,
      hangupRate: hangupRate,
      callCount: callCount
    }),
    muteHttpExceptions: true
  });
}

// For RPC Alerts
function sendRPCAlert(targetName, rpc, threshold) {
  UrlFetchApp.fetch('http://localhost:8888/alert', {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify({
      type: 'rpc',
      target: targetName,
      rpc: rpc,
      threshold: threshold
    }),
    muteHttpExceptions: true
  });
}
